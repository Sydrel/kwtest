import { serve } from 'https://deno.land/std@0.168.0/http/server.ts'; import Stripe from 'https://esm.sh/stripe@12.0.0?target=deno'; const stripe = new Stripe(Deno.env.get('STRIPE_SECRET_KEY') || '', { apiVersion: '2022-11-15', }); const corsHeaders = { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type', }; serve(async (req) => { if (req.method === 'OPTIONS') { return new Response('ok', { headers: corsHeaders }); } try { const { amount, addonsAmount, setupFee, isSubscription, packageName, clientName, email, phone } = await req.json(); let customer; const existingCustomers = await stripe.customers.list({ email: email, limit: 1 }); if (existingCustomers.data.length > 0) { customer = existingCustomers.data[0]; } else { customer = await stripe.customers.create({ name: clientName, email: email, phone: phone || undefined, metadata: { business: 'KW Commercial Fleet Washing & Mobile Detailing' } }); } if (isSubscription) { // 1. Create the recurring monthly price for the subscription plan itself const price = await stripe.prices.create({ unit_amount: amount, // Monthly recurring cost in cents currency: 'usd', recurring: { interval: 'month' }, product_data: { name: `${packageName} (Monthly Subscription)`, }, }); // 2. Bundle one-time day-one charges ($50 Setup Fee + Any Selected Add-ons) const oneTimeInvoiceItems = []; if (setupFee && setupFee > 0) { oneTimeInvoiceItems.push({ price_data: { currency: 'usd', unit_amount: setupFee, // 5000 for $50.00 setup fee product_data: { name: 'Non-refundable Setup Fee' }, }, }); } if (addonsAmount && addonsAmount > 0) { oneTimeInvoiceItems.push({ price_data: { currency: 'usd', unit_amount: addonsAmount, // One-time add-on total in cents product_data: { name: 'Selected One-Time Add-On Services' }, }, }); } // 3. Create the subscription with setup fee & add-ons added ONLY to the first invoice const subscription = await stripe.subscriptions.create({ customer: customer.id, items: [{ price: price.id }], add_invoice_items: oneTimeInvoiceItems, // Automatically bills $50 setup fee & add-ons on day 1 only payment_behavior: 'default_incomplete', payment_settings: { save_default_payment_method: 'on_subscription' }, expand: ['latest_invoice.payment_intent'], metadata: { package_name: packageName, is_subscription: 'true', }, }); const invoice = subscription.latest_invoice as Stripe.Invoice; const paymentIntent = invoice.payment_intent as Stripe.PaymentIntent; return new Response( JSON.stringify({ clientSecret: paymentIntent.client_secret, subscriptionId: subscription.id, customerId: customer.id }), { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200 } ); } else { // Handle One-Time Bookings (Base Plan + $50 Setup Fee + Add-ons all bundled together) const paymentIntent = await stripe.paymentIntents.create({ amount: amount + (setupFee || 0) + (addonsAmount || 0), currency: 'usd', customer: customer.id, metadata: { is_subscription: 'false', package_name: packageName, billing_cycle: 'one_time' }, automatic_payment_methods: { enabled: true } }); return new Response( JSON.stringify({ clientSecret: paymentIntent.client_secret, customerId: customer.id }), { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200 } ); } } catch (error) { return new Response( JSON.stringify({ error: error.message }), { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 400 } ); } });
